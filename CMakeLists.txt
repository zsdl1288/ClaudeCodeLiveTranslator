cmake_minimum_required(VERSION 3.20)
project(ClaudeLiveTranslator VERSION 1.3.1 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This application uses Windows UI Automation and WinHTTP.")
endif()
# Use the bundled resource compiler for the portable MinGW build. MSVC keeps rc.exe.
if(NOT MSVC AND NOT CMAKE_RC_COMPILER)
  set(CCLIVE_BUNDLED_RC "${CMAKE_CURRENT_SOURCE_DIR}/toolchain/llvm-mingw-20260908-ucrt-x86_64/bin/llvm-windres.exe")
  if(EXISTS "${CCLIVE_BUNDLED_RC}")
    set(CMAKE_RC_COMPILER "${CCLIVE_BUNDLED_RC}" CACHE FILEPATH "Windows resource compiler")
  endif()
endif()
enable_language(RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/utf-8;/W4>")
endif()
set(CCLIVE_SYSTEM_LIBS uiautomationcore ole32 oleaut32 uuid winhttp comctl32 comdlg32 shell32 shlwapi gdi32 user32 advapi32 ws2_32 crypt32)
add_library(cclive_core STATIC src/core.cpp src/chatgpt.cpp src/codex_cli.cpp src/task_cache.cpp src/text_layout.cpp)
target_include_directories(cclive_core PUBLIC include third_party)
target_compile_definitions(cclive_core PUBLIC UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_link_libraries(cclive_core PUBLIC ${CCLIVE_SYSTEM_LIBS})
add_executable(ClaudeTranslator WIN32 src/main.cpp src/reader.cpp resources/app.rc)
target_include_directories(ClaudeTranslator PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/resources")
set_source_files_properties(resources/app.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/app.ico;${CMAKE_CURRENT_SOURCE_DIR}/include/resource.h")
target_link_libraries(ClaudeTranslator PRIVATE cclive_core)
set_target_properties(ClaudeTranslator PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/build-cpp")
if(MSVC)
  # All targets share the static runtime and UTF-8 source setting above.
else()
  target_compile_options(cclive_core PUBLIC "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wno-unknown-pragmas>")
  target_link_options(ClaudeTranslator PRIVATE -static -municode)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE cclive_core)
  if(NOT MSVC)
    target_link_options(core_tests PRIVATE -static)
  endif()
  add_test(NAME core_tests COMMAND core_tests WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
  add_executable(chatgpt_tests tests/chatgpt_tests.cpp src/chatgpt.cpp src/codex_cli.cpp src/core.cpp)
  target_include_directories(chatgpt_tests PRIVATE include third_party)
  target_compile_definitions(chatgpt_tests PRIVATE CCLIVE_CHATGPT_TESTING UNICODE _UNICODE _WIN32_WINNT=0x0A00)
  target_link_libraries(chatgpt_tests PRIVATE ${CCLIVE_SYSTEM_LIBS})
  if(NOT MSVC)
    target_link_options(chatgpt_tests PRIVATE -static)
  endif()
  add_test(NAME chatgpt_tests COMMAND chatgpt_tests WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
  add_executable(codex_cli_tests tests/codex_cli_tests.cpp src/codex_cli.cpp src/core.cpp src/chatgpt.cpp)
  target_include_directories(codex_cli_tests PRIVATE include third_party)
  target_compile_definitions(codex_cli_tests PRIVATE CCLIVE_CODEX_CLI_TESTING UNICODE _UNICODE _WIN32_WINNT=0x0A00)
  target_link_libraries(codex_cli_tests PRIVATE ${CCLIVE_SYSTEM_LIBS})
  if(NOT MSVC)
    target_link_options(codex_cli_tests PRIVATE -static -municode)
  endif()
  add_test(NAME codex_cli_tests COMMAND codex_cli_tests WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
  add_executable(task_cache_tests tests/task_cache_tests.cpp)
  target_link_libraries(task_cache_tests PRIVATE cclive_core)
  if(NOT MSVC)
    target_link_options(task_cache_tests PRIVATE -static)
  endif()
  add_test(NAME task_cache_tests COMMAND task_cache_tests WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
  add_executable(text_layout_tests tests/text_layout_tests.cpp)
  target_link_libraries(text_layout_tests PRIVATE cclive_core)
  if(NOT MSVC)
    target_link_options(text_layout_tests PRIVATE -static)
  endif()
  add_test(NAME text_layout_tests COMMAND text_layout_tests WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
  add_test(NAME native_gui_selftest COMMAND ClaudeTranslator --self-test --report "${CMAKE_BINARY_DIR}/native-gui-check.json" WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
  set_tests_properties(native_gui_selftest PROPERTIES TIMEOUT 20)
  if(EXISTS "${CMAKE_SOURCE_DIR}/tests/fixture.cpp")
    add_executable(CCLiveFixture WIN32 tests/fixture.cpp)
    target_compile_definitions(CCLiveFixture PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
    target_link_libraries(CCLiveFixture PRIVATE ${CCLIVE_SYSTEM_LIBS})
    if(NOT MSVC)
      target_link_options(CCLiveFixture PRIVATE -static -municode)
    endif()
  endif()
  if(EXISTS "${CMAKE_SOURCE_DIR}/tests/reader_smoke.cpp")
    add_executable(reader_smoke tests/reader_smoke.cpp)
    target_link_libraries(reader_smoke PRIVATE cclive_core)
    if(NOT MSVC)
      target_link_options(reader_smoke PRIVATE -static -municode)
    endif()
    add_test(NAME reader_smoke COMMAND reader_smoke $<TARGET_FILE:CCLiveFixture> $<TARGET_FILE:ClaudeTranslator> WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
  endif()
endif()
install(TARGETS ClaudeTranslator RUNTIME DESTINATION .)
install(FILES LICENSE README.md CHANGELOG.md DESTINATION .)
install(FILES third_party/nlohmann/LICENSE DESTINATION licenses RENAME nlohmann-json-LICENSE)
install(FILES third_party/llvm-LICENSE DESTINATION licenses)

install(FILES third_party/desktop-translator-LICENSE DESTINATION licenses)
install(DIRECTORY third_party/mingw/ DESTINATION licenses/mingw)
